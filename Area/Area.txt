*Before we start this chapter Note that Area 0 known as back bone.
*The condition for being able to have multi area is that we "MUST" have at least one router interface on each side ex: one int on area 0 and one in area 10 (one in backbone on in target area) 
* ASBR can be anywhere it doesn't matter if it's in backbone or other areas
======R1 Configs============
en 
conf t
hostname R1
int g 0/0 
no sh 
ip add 10.10.12.1 255.255.255.248
int g 0/1
no sh
ip add 10.10.14.1 255.255.255.248
int g 0/2
no sh
ip add 10.10.13.1 255.255.255.248
exit
ip ospf 1 
router-id 1.1.1.1
network 10.0.0.0 0.255.255.255 area 0
exit
do wr

======R5 Configs============
en 
conf t
hostname R5
int g 0/0 
no sh 
ip add 10.10.25.1 255.255.255.248
int loopback 1
ip add 192.168.0.1 255.255.255.248
ip ospf network point-to-point
int loopback2
ip add 192.168.1.1 255.255.255.248
ip ospf network point-to-point
int loopback3 
ip add 192.168.2.1 255.255.255.248
ip ospf network point-to-point
int loopback 4
ip add 192.168.3.1 255.255.255.248
ip ospf network point-to-point
exit

network 10.10.25.0 0.0.0.255 area 10
