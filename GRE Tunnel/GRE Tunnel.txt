# In this scenario we are going to run GRE tunnel.
# Note that GRE Tunnel isn't that safe due to being clear text while transferring data 

===========HQ-CE1 Configs================
en 
conf t
hostname HQ_CE1
int g 0/0 
no sh
ip add 10.10.10.10.1 255.255.255.0
int g 0/1
no sh
ip add 100.100.100.1 255.255.255.0
interface Tunnel0
ip address 172.16.12.1 255.255.255.252
tunnel source G 0/1
tunnel destination 200.200.200.1
tunnel mode gre ip
no shutdown
exit
ip route 10.10.20.0 255.255.255.0 172.16.12.2
------------------------------------------------
Tunnel0 = 172.16.12.1
Source  = 100.100.100.1
Destination = 200.200.200.1
------------------------------------------------
